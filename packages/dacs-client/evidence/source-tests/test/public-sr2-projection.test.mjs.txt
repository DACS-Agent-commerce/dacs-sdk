test('realistic RPC accepting consumer: approved public signed CREATE reaches seller HTTP, fresh native receipt verifies, exact fixture commitment reads back independently and payment stays held', async (t) => {
  const w = await publicProjectionFlow(t);
  const out = await w.run(), p = w.op().state.protocol, jobId = w.op().terms.protocol.jobId;
  assert.equal(out.protocol.stage, 'seller-committed', JSON.stringify(p.attempts));
  assert.equal(out.status, 'agreement_pending'); assert.equal(out.protocol.limits.payment, 'held');
  assert.equal(w.h.sellerChainUrl, w.h.chain.url, 'no idealized RPC front');
  const signedProjection = p.agreement.signedProjection;
  assert.equal(JSON.stringify(signedProjection), JSON.stringify(w.h.chain.wire.record.accepted[0]), 'exact signed bytes retained and conveyed');
  const executed = await post(w.h.chain.url, { method: 'nodeCall', params: [{ message: 'getTxByHash', data: { hash: signedProjection.hash } }] });
  assert.equal(typeof executed.response.content.nonce, 'string');
  assert.equal(sha256Hex(JSON.stringify(signedProjection.content)), signedProjection.hash);
  assert.notEqual(sha256Hex(JSON.stringify(executed.response.content)), signedProjection.hash, 'realistic normalization witnessed');
  const journal = createInMemoryDemosSr2Journal();
  const adapter = createDemosSr2Adapter({ rpcUrl: w.h.chain.url, journal });
  const reader = createSellerSr2AgreementAnchor({ reader: adapter });
  const input = { logicalAddress: fixedPriceAgreementLogicalAddress(jobId), writer: p.buyerBundle.presentedBy, agreementHash: p.agreement.hash, signedProjection };
  const read = await reader.readAgreement(input);
  assert.equal(read.disposition, 'present', JSON.stringify(read));
  assert.equal(read.native.nativeAddress, p.agreement.anchorReceipt.nativeAddress);
  assert.equal(await adapter.verifyReceipt({ receipt: p.agreement.anchorReceipt, logicalAddress: input.logicalAddress, artifact: p.agreement.artifact, expectedWriter: input.writer, artifactKind: 'Agreement', signedProjection }), 'valid');
  assert.deepEqual(journal.records(), [], 'third party fabricated or retained no writer state');
  const committed = await readCommitment({ agreement: p.agreement.artifact, listing: w.seller.listing, anchors: createFixtureAnchorClient({ url: w.h.anchors.url }), now: Date.now });
  assert.equal(committed.status, 'verified'); assert.equal(committed.contentHash, out.protocol.commitment.contentHash);
  assert.equal(committed.signer, w.seller.wallet.claim);
  assert.equal(contentHash(w.h.anchors.records.get(finalityCommitmentAddress(jobId)).artifact), committed.contentHash);
  assert.equal(w.seller.anchorWrites.length, 1); assert.equal(w.h.chain.wire.record.broadcastAttempts.length, 1);
  assert.equal(w.approvals.filter((r) => r.content?.kind === 'dacs-storage-anchor-create').length, 1);
  for (const part of ['challenge', 'handoff']) assert.equal((await post(`${w.h.origin}/portal/api/purchase/${w.id}/${part}`, { token: w.token, address: w.address })).error, 'agreement_payment_held');
  assert.ok(!w.seller.requests.includes('/dacs/v1/payments'));
  t.diagnostic(JSON.stringify({ stage: out.protocol.stage, nativeAddress: read.native.nativeAddress, createHash: signedProjection.hash, commitmentHash: committed.contentHash, commitmentWrites: w.seller.anchorWrites.length, buyerBroadcasts: 1, payment: 'held', rpc: 'realistic-normalizing', transport: 'legacy-loopback' }));
});

test('realistic RPC lifecycle: durable seller restart retains public proof, reauthenticates fresh observations and resumes commitment without a buyer CREATE or commitment resend', async (t) => {
  const w = await publicProjectionFlow(t, { durable: true });
  const out = await w.run(), jobId = w.op().terms.protocol.jobId;
  assert.equal(out.protocol.stage, 'seller-committed', JSON.stringify(out.protocol.lastAttempt));
  await w.seller.restart();
  w.h.chain.faults.unavailable = true;
  assert.equal((await post(`${w.seller.origin}/dacs/v1/commitments`, { jobId })).httpStatus, 503, 'retained proof still needs fresh observation');
  w.h.chain.faults.unavailable = false;
  const resumed = await post(`${w.seller.origin}/dacs/v1/commitments`, { jobId });
  assert.equal(resumed.status, 'committed', JSON.stringify(resumed)); assert.equal(resumed.resumed, true);
  assert.equal(resumed.agreementAnchor.nativeAddress, w.op().state.protocol.agreement.anchorReceipt.nativeAddress);
  assert.equal((await w.verify()).protocol.commitment.contentHash, out.protocol.commitment.contentHash);
  assert.deepEqual(w.seller.anchorWrites, [finalityCommitmentAddress(jobId)], 'no commitment effect resend');
  assert.equal(w.h.chain.wire.record.broadcastAttempts.length, 1);
  t.diagnostic(JSON.stringify({ durableRestart: w.seller.restarts(), commitmentHash: out.protocol.commitment.contentHash, commitmentWrites: 1, buyerBroadcasts: 1, payment: 'held' }));
});

test('authenticated HTTP accepts the whole public projection; lost acknowledgement reconciles on durable restart without performing commitment again', async (t) => {
  const w = await publicProjectionFlow(t, { capability: false, durable: true });
  assert.equal((await w.run()).protocol.stage, 'agreement-anchored');
  const p = w.op().state.protocol, jobId = w.op().terms.protocol.jobId;
  const signedProjection = w.memory.snapshot().find(([key, r]) => key.startsWith('sr2/adapter/') && r.transactionHash === p.agreement.anchorReceipt.transactionRef.value)[1].authorizedTransaction;
  const dir = realpathSync(mkdtempSync(join(tmpdir(), 'dacs-public-peer-')));
  let store, handler;
  const server = http.createServer((req, res) => handler(req, res));
  await new Promise((resolve) => server.listen(0, '127.0.0.1', resolve));
  t.after(async () => { server.closeAllConnections(); await new Promise((resolve) => server.close(resolve)); store.close(); rmSync(dir, { recursive: true }); });
  const serving = () => {
    store = createSellerTransportStore({ path: join(dir, 'peer.sqlite') });
    const authentication = createSellerTransportServerAuth({ sellerSigner: { claim: w.seller.wallet.claim, sign: w.seller.wallet.signBytes }, listingPin: listingPin(w.seller.listing), store });
    handler = createSellerHttpHandler(w.seller.agent, { authentication });
  };
  serving();
  const clientAuth = createSellerTransportClientAuth({ buyerSigner: { claim: p.buyerBundle.presentedBy, sign: (bytes) => w.sign(bytes.toString('utf8')) }, seller: w.seller.wallet.claim, resolveJob: () => ({ listingPin: listingPin(w.seller.listing), agreementHash: p.agreement.hash }) });
  let signedBody;
  const authentication = { ...clientAuth, request: async (path, body, text) => { const result = await clientAuth.request(path, body, text); signedBody = { body, text, hash: result.payload.bodyHash }; return result; } };
  const client = createSellerAgentClient({ url: `http://127.0.0.1:${server.address().port}`, authentication });
  assert.equal((await client.propose(w.seller.proposal)).disposition, 'complete', 'normal authenticated admission precedes commitment');
  const finish = store.finish;
  store.finish = (...args) => { if (args[1] === 200) throw new Error('test-owned lost response acknowledgement'); return finish(...args); };
  await assert.rejects(client.commit(jobId, { signedProjection }), SellerUnavailableError);
  assert.equal(signedBody.hash, sha256Hex(signedBody.text));
  assert.deepEqual(signedBody.body, { jobId, signedProjection }, 'full optional projection in authenticated body');
  assert.equal(w.seller.agent.stageOf(jobId), 'committed');
  store.close(); await w.seller.restart(); serving();
  const resumed = await client.commit(jobId, { signedProjection });
  assert.equal(resumed.status, 'committed'); assert.equal(resumed.resumed, true);
  assert.equal(resumed.agreementAnchor.nativeAddress, p.agreement.anchorReceipt.nativeAddress);
  assert.equal(w.seller.anchorWrites.length, 1, 'read-only transport reconciliation performs no effect');
  assert.equal(w.h.chain.wire.record.broadcastAttempts.length, 1);
  const read = await readCommitment({ agreement: p.agreement.artifact, listing: w.seller.listing, anchors: createFixtureAnchorClient({ url: w.h.anchors.url }), now: Date.now });
  assert.equal(read.status, 'verified');
  t.diagnostic(JSON.stringify({ authenticatedProjectionBodyHash: signedBody.hash, reconciliation: 'read-only-after-lost-ack', commitmentHash: read.contentHash, commitmentWrites: 1, buyerBroadcasts: 1, appTransport: 'legacy-loopback; authenticated client tested separately', payment: 'held' }));
});

test('public projection accepting reader: absent Agreement and unavailable fresh chain remain held; no writer plan is populated', async (t) => {
  const w = await publicProjectionFlow(t); await w.run();
  const p = w.op().state.protocol;
  const reader = createSellerSr2AgreementAnchor({ rpcUrl: w.h.chain.url });
  const input = { logicalAddress: fixedPriceAgreementLogicalAddress(w.op().terms.protocol.jobId), writer: p.buyerBundle.presentedBy, agreementHash: p.agreement.hash, signedProjection: p.agreement.signedProjection };
  const { signedProjection, ...withoutProjection } = input;
  const absentInput = await reader.readAgreement(withoutProjection);
  assert.equal(absentInput.disposition, 'rejected', 'absent public bytes keep the original realistic RPC hold');
  assert.match(absentInput.reason, /transaction lacks the exact writer authorization/);
  assert.equal((await reader.readAgreement({ ...input, logicalAddress: fixedPriceAgreementLogicalAddress('01M45JZZZZZZZZZZZZZZZZZZZZ') })).disposition, 'absent');
  w.h.chain.faults.unavailable = true;
  assert.equal((await reader.readAgreement(input)).disposition, 'indeterminate');
  w.h.chain.faults.unavailable = false;
  assert.equal((await reader.readAgreement(input)).disposition, 'present');
  assert.deepEqual(sellerSr2CommitmentConfig({ commitmentSubstrate: 'fixture' }), { commitmentSubstrate: 'fixture' });
  assert.deepEqual(sellerSr2CommitmentConfig({ commitmentSubstrate: 'fixture', publicProjection: true }), { commitmentSubstrate: 'fixture', publicProjection: true });
});
